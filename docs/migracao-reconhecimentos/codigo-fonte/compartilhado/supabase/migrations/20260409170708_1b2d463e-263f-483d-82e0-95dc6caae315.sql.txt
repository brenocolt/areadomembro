
CREATE TABLE public.projetos_entradas (
  id UUID NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
  nome TEXT NOT NULL,
  ordem INTEGER NOT NULL DEFAULT 0,
  ativo BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now()
);

ALTER TABLE public.projetos_entradas ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow all for authenticated" ON public.projetos_entradas FOR ALL TO authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Allow read for anon" ON public.projetos_entradas FOR SELECT TO anon USING (true);

ALTER TABLE public.categorias_entradas ADD COLUMN projeto_id UUID REFERENCES public.projetos_entradas(id) ON DELETE CASCADE;
