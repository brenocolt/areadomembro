
-- Limpar dados existentes
DELETE FROM lancamentos;
DELETE FROM projecoes;

-- Tabela de categorias de entradas (centros de custo de entradas na UI)
CREATE TABLE public.categorias_entradas (
  id UUID NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
  nome TEXT NOT NULL,
  ativo BOOLEAN NOT NULL DEFAULT true,
  ordem INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now()
);

ALTER TABLE public.categorias_entradas ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow all for authenticated" ON public.categorias_entradas FOR ALL TO authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Allow read for anon" ON public.categorias_entradas FOR SELECT TO anon USING (true);

-- Tabela de centros de custo de entradas (categorias de entradas na UI)
CREATE TABLE public.centros_custo_entradas (
  id UUID NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
  categoria_entrada_id UUID NOT NULL REFERENCES public.categorias_entradas(id) ON DELETE CASCADE,
  nome TEXT NOT NULL,
  ativo BOOLEAN NOT NULL DEFAULT true,
  ordem INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now()
);

ALTER TABLE public.centros_custo_entradas ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow all for authenticated" ON public.centros_custo_entradas FOR ALL TO authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Allow read for anon" ON public.centros_custo_entradas FOR SELECT TO anon USING (true);

-- Tabela de projeções de entradas
CREATE TABLE public.projecoes_entradas (
  id UUID NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
  ano INTEGER NOT NULL,
  mes INTEGER NOT NULL,
  categoria_entrada_id UUID NOT NULL REFERENCES public.categorias_entradas(id) ON DELETE CASCADE,
  centro_custo_entrada_id UUID REFERENCES public.centros_custo_entradas(id) ON DELETE SET NULL,
  valor_estimado NUMERIC NOT NULL DEFAULT 0,
  updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
  updated_by TEXT
);

ALTER TABLE public.projecoes_entradas ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow all for authenticated" ON public.projecoes_entradas FOR ALL TO authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Allow all for anon" ON public.projecoes_entradas FOR ALL TO anon USING (true) WITH CHECK (true);
