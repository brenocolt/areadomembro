
CREATE TABLE public.inadimplentes (
  id UUID NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
  projeto TEXT NOT NULL,
  mes_inadimplencia INTEGER NOT NULL CHECK (mes_inadimplencia BETWEEN 1 AND 12),
  ano INTEGER NOT NULL,
  parcelas_inadimplentes INTEGER NOT NULL DEFAULT 1 CHECK (parcelas_inadimplentes > 0),
  observacoes TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

GRANT SELECT, INSERT, UPDATE, DELETE ON public.inadimplentes TO authenticated, anon;
GRANT ALL ON public.inadimplentes TO service_role;

ALTER TABLE public.inadimplentes ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public read inadimplentes" ON public.inadimplentes FOR SELECT USING (true);
CREATE POLICY "Public insert inadimplentes" ON public.inadimplentes FOR INSERT WITH CHECK (true);
CREATE POLICY "Public update inadimplentes" ON public.inadimplentes FOR UPDATE USING (true);
CREATE POLICY "Public delete inadimplentes" ON public.inadimplentes FOR DELETE USING (true);

CREATE OR REPLACE FUNCTION public.update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN NEW.updated_at = now(); RETURN NEW; END;
$$ LANGUAGE plpgsql SET search_path = public;

CREATE TRIGGER update_inadimplentes_updated_at
  BEFORE UPDATE ON public.inadimplentes
  FOR EACH ROW EXECUTE FUNCTION public.update_updated_at_column();
