import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { supabase } from '@/integrations/supabase/client';

export function useCategorias() {
  return useQuery({
    queryKey: ['categorias'],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('categorias')
        .select('*')
        .order('ordem');
      if (error) throw error;
      return data;
    },
  });
}

export function useLancamentos(mes?: number, ano?: number) {
  return useQuery({
    queryKey: ['lancamentos', mes, ano],
    queryFn: async () => {
      let query = supabase
        .from('lancamentos')
        .select('*')
        .order('data', { ascending: false });
      if (mes !== undefined) query = query.eq('mes_referencia', mes);
      if (ano !== undefined) query = query.eq('ano_referencia', ano);
      const { data, error } = await query;
      if (error) throw error;
      return data;
    },
  });
}

export function useProjecoes(ano: number) {
  return useQuery({
    queryKey: ['projecoes', ano],
    queryFn: async () => {
      const pageSize = 1000;
      let from = 0;
      const all: any[] = [];
      while (true) {
        const { data, error } = await supabase
          .from('projecoes')
          .select('*, categorias(nome), centros_custo(nome)')
          .eq('ano', ano)
          .range(from, from + pageSize - 1);
        if (error) throw error;
        if (!data || data.length === 0) break;
        all.push(...data);
        if (data.length < pageSize) break;
        from += pageSize;
      }
      return all;
    },
  });
}

export function useSaldoInicial(ano: number) {
  return useQuery({
    queryKey: ['saldo_inicial', ano],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('saldos_iniciais')
        .select('*')
        .eq('ano', ano)
        .maybeSingle();
      if (error) throw error;
      return data;
    },
  });
}

export function useUpsertSaldoInicial() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async ({ ano, valor }: { ano: number; valor: number }) => {
      const { data: existing } = await supabase
        .from('saldos_iniciais')
        .select('id')
        .eq('ano', ano)
        .maybeSingle();
      if (existing) {
        const { error } = await supabase.from('saldos_iniciais').update({ valor }).eq('id', existing.id);
        if (error) throw error;
      } else {
        const { error } = await supabase.from('saldos_iniciais').insert({ ano, valor });
        if (error) throw error;
      }
    },
    onSuccess: (_, { ano }) => {
      queryClient.invalidateQueries({ queryKey: ['saldo_inicial', ano] });
    },
  });
}

export function useLancamentosAno(ano: number) {
  return useQuery({
    queryKey: ['lancamentos_ano', ano],
    queryFn: async () => {
      const PAGE = 1000;
      const all: any[] = [];
      let from = 0;
      // Paginação manual para contornar o teto padrão do PostgREST (1000 linhas).
      while (true) {
        const { data, error } = await supabase
          .from('lancamentos')
          .select('tipo, valor, mes_referencia, categoria_id, centro_custo_id, categoria_entrada_id, centro_custo_entrada_id')
          .eq('ano_referencia', ano)
          .order('id', { ascending: true })
          .range(from, from + PAGE - 1);
        if (error) throw error;
        if (!data || data.length === 0) break;
        all.push(...data);
        if (data.length < PAGE) break;
        from += PAGE;
      }
      return all;
    },
  });
}
