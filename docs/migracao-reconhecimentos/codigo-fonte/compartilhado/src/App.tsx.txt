import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { BrowserRouter, Route, Routes } from "react-router-dom";
import { Toaster as Sonner } from "@/components/ui/sonner";
import { Toaster } from "@/components/ui/toaster";
import { TooltipProvider } from "@/components/ui/tooltip";
import { AppLayout } from "@/components/layout/AppLayout";
import { ProtectedRoute } from "@/components/ProtectedRoute";
import { AdminRoute } from "@/components/AdminRoute";
import Login from "./pages/Login";
import Dashboard from "./pages/Dashboard";
import FluxoMensal from "./pages/FluxoMensal";
import ProjecoesVisualizacao from "./pages/ProjecoesVisualizacao";
import ProjecoesEditar from "./pages/ProjecoesEditar";
import ProjecoesCategorias from "./pages/ProjecoesCategorias";
import ProjecoesEntradasCategorias from "./pages/ProjecoesEntradasCategorias";
import ProjecoesEntradasVisualizacao from "./pages/ProjecoesEntradasVisualizacao";
import ProjecoesEntradasEditar from "./pages/ProjecoesEntradasEditar";
import Eventos from "./pages/Eventos";
import Assistente from "./pages/Assistente";
import AssembleiaGeral from "./pages/AssembleiaGeral";
import ContasSolicitacoes from "./pages/ContasSolicitacoes";
import AcompanhamentoGastos from "./pages/AcompanhamentoGastos";
import Reconhecimentos from "./pages/Reconhecimentos";
import NotFound from "./pages/NotFound";

const queryClient = new QueryClient();

const App = () => (
  <QueryClientProvider client={queryClient}>
    <TooltipProvider>
      <Toaster />
      <Sonner />
      <BrowserRouter>
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route
            path="*"
            element={
              <ProtectedRoute>
                <AppLayout>
                  <Routes>
                    <Route path="/" element={<Dashboard />} />
                    <Route path="/fluxo" element={<FluxoMensal />} />
                    <Route path="/projecoes" element={<ProjecoesVisualizacao />} />
                    <Route path="/projecoes/editar" element={<ProjecoesEditar />} />
                    <Route path="/projecoes/categorias" element={<ProjecoesCategorias />} />
                    <Route path="/projecoes-entradas" element={<ProjecoesEntradasVisualizacao />} />
                    <Route path="/projecoes-entradas/editar" element={<ProjecoesEntradasEditar />} />
                    <Route path="/projecoes-entradas/categorias" element={<ProjecoesEntradasCategorias />} />
                    <Route path="/eventos" element={<Eventos />} />
                    <Route path="/assembleia" element={<AssembleiaGeral />} />
                    <Route path="/assistente" element={<Assistente />} />
                    <Route path="/gastos" element={<AcompanhamentoGastos />} />
                    <Route path="/reconhecimentos" element={<Reconhecimentos />} />
                    <Route path="/contas" element={<AdminRoute><ContasSolicitacoes /></AdminRoute>} />
                    <Route path="*" element={<NotFound />} />
                  </Routes>
                </AppLayout>
              </ProtectedRoute>
            }
          />
        </Routes>
      </BrowserRouter>
    </TooltipProvider>
  </QueryClientProvider>
);

export default App;
