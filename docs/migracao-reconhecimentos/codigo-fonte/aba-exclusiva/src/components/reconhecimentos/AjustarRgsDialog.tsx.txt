import { useMemo, useState } from 'react';
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription } from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { SearchableSelect } from '@/components/ui/searchable-select';
import { toast } from 'sonner';
import { supabase } from '@/integrations/supabase/client';
import { useQueryClient } from '@tanstack/react-query';
import { useReconhecimentos } from '@/hooks/useBonificacoes';
import { sincronizarBonificacoes, currentUser } from '@/lib/bonificacoes-engine';

export const OBS_RG_ANTERIOR = 'RG adiantada registrada antes do uso do sistema';

interface Props {
  open: boolean;
  onOpenChange: (open: boolean) => void;
}

/**
 * Ajusta a contagem de RGs adiantadas anteriores ao sistema de um projeto,
 * inserindo ou removendo registros históricos e recalculando as bonificações.
 */
export function AjustarRgsDialog({ open, onOpenChange }: Props) {
  const queryClient = useQueryClient();
  const { data: reconhecimentos = [] } = useReconhecimentos();

  const [projetoId, setProjetoId] = useState('');
  const [quantidade, setQuantidade] = useState('');
  const [saving, setSaving] = useState(false);

  const projetos = useMemo(() => {
    const mapa = new Map<string, string>();
    (reconhecimentos as any[]).forEach((r) => {
      if (r.projeto_id) mapa.set(r.projeto_id, r.projeto_nome || 'Projeto');
    });
    return [...mapa.entries()]
      .map(([id, nome]) => ({ id, nome }))
      .sort((a, b) => a.nome.localeCompare(b.nome, 'pt-BR'));
  }, [reconhecimentos]);

  const recsProjeto = useMemo(
    () => (reconhecimentos as any[]).filter((r) => r.projeto_id === projetoId),
    [reconhecimentos, projetoId],
  );
  const anteriores = useMemo(
    () => recsProjeto.filter((r) => r.observacao === OBS_RG_ANTERIOR),
    [recsProjeto],
  );
  const totalAdiantados = recsProjeto.filter(
    (r) => r.adiantado || r.rg_adiantada || r.projeto_adiantado,
  ).length;

  const reset = () => {
    setProjetoId('');
    setQuantidade('');
  };

  const salvar = async () => {
    if (!projetoId) return toast.error('Selecione o projeto');
    const novaQtd = Math.max(0, Math.floor(Number(quantidade) || 0));
    const diff = novaQtd - anteriores.length;
    if (diff === 0) {
      toast.info('A quantidade informada é igual à atual');
      return;
    }

    setSaving(true);
    try {
      if (diff > 0) {
        const base = recsProjeto[0];
        const primeiraData =
          recsProjeto.map((r) => r.data).sort()[0] || new Date().toISOString().slice(0, 10);
        const rows = Array.from({ length: diff }, () => ({
          tipo_id: base.tipo_id,
          data: primeiraData,
          projeto_id: projetoId,
          projeto_nome: base.projeto_nome || '',
          adiantado: true,
          rg_adiantada: false,
          projeto_adiantado: false,
          nucleo_impacto: null,
          observacao: OBS_RG_ANTERIOR,
          criado_por: currentUser(),
        }));
        const { error } = await supabase.from('reconhecimentos').insert(rows);
        if (error) throw error;
      } else {
        const remover = [...anteriores]
          .sort((a, b) => String(b.created_at).localeCompare(String(a.created_at)))
          .slice(0, -diff);
        const { error } = await supabase
          .from('reconhecimentos')
          .delete()
          .in('id', remover.map((r) => r.id));
        if (error) throw error;
      }

      await sincronizarBonificacoes();
      queryClient.invalidateQueries({ queryKey: ['reconhecimentos'] });
      queryClient.invalidateQueries({ queryKey: ['bonificacoes'] });
      toast.success('Contagem de RGs ajustada!');
      reset();
      onOpenChange(false);
    } catch (e: any) {
      toast.error(e.message || 'Erro ao ajustar');
    } finally {
      setSaving(false);
    }
  };

  return (
    <Dialog open={open} onOpenChange={(o) => { onOpenChange(o); if (!o) reset(); }}>
      <DialogContent className="max-w-lg">
        <DialogHeader>
          <DialogTitle>Ajustar RGs anteriores ao sistema</DialogTitle>
          <DialogDescription>
            Corrija quantas RGs adiantadas o projeto já tinha antes do uso do sistema. A contagem e as bonificações são recalculadas automaticamente.
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-4">
          <div className="space-y-1.5">
            <Label>Projeto</Label>
            <SearchableSelect
              value={projetoId}
              onValueChange={(v) => { setProjetoId(v); setQuantidade(''); }}
              options={projetos.map((p) => ({ value: p.id, label: p.nome }))}
              placeholder="Pesquisar ou selecionar projeto..."
              searchPlaceholder="Pesquisar pelo nome..."
              emptyText="Nenhum projeto com reconhecimentos"
            />
          </div>

          {projetoId && (
            <>
              <div className="rounded-md border border-border bg-muted/30 p-3 text-sm">
                <p>Total de reconhecimentos do projeto: <strong>{recsProjeto.length}</strong></p>
                <p>Reconhecimentos adiantados: <strong>{totalAdiantados}</strong></p>
                <p>RGs anteriores ao sistema (atual): <strong>{anteriores.length}</strong></p>
              </div>

              <div className="space-y-1.5">
                <Label>Nova quantidade de RGs anteriores ao sistema</Label>
                <Input
                  type="number"
                  min={0}
                  value={quantidade}
                  onChange={(e) => setQuantidade(e.target.value)}
                  placeholder={String(anteriores.length)}
                />
                <p className="text-xs text-muted-foreground">
                  Bonificações já geradas são mantidas no histórico. Novas bonificações são geradas automaticamente se a contagem atingir alguma regra.
                </p>
              </div>
            </>
          )}
        </div>

        <DialogFooter>
          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancelar</Button>
          <Button onClick={salvar} disabled={saving || !projetoId}>{saving ? 'Salvando...' : 'Salvar ajuste'}</Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}
