
-- Tipos de reconhecimento
CREATE TABLE public.reconhecimento_tipos (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  nome text NOT NULL,
  escopo text NOT NULL DEFAULT 'pessoa', -- 'projeto' | 'pessoa'
  ativo boolean NOT NULL DEFAULT true,
  ordem integer NOT NULL DEFAULT 0,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.reconhecimento_tipos TO anon, authenticated;
GRANT ALL ON public.reconhecimento_tipos TO service_role;
ALTER TABLE public.reconhecimento_tipos ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all reconhecimento_tipos" ON public.reconhecimento_tipos FOR ALL USING (true) WITH CHECK (true);

-- Regras de bonificacao
CREATE TABLE public.regras_bonificacao (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  nome text NOT NULL,
  tipo_id uuid REFERENCES public.reconhecimento_tipos(id) ON DELETE CASCADE,
  criterio text NOT NULL DEFAULT 'por_reconhecimento', -- 'simbolico' | 'por_reconhecimento' | 'acumulado'
  quantidade_necessaria integer NOT NULL DEFAULT 1,
  ciclo integer NOT NULL DEFAULT 0, -- 0 = nao repete; n = repete a cada n
  requer_rg_adiantada boolean NOT NULL DEFAULT false,
  requer_projeto_adiantado boolean NOT NULL DEFAULT false,
  tipo_premiacao text NOT NULL DEFAULT 'financeira', -- 'financeira' | 'premio' | 'simbolica'
  valor numeric NOT NULL DEFAULT 0,
  descricao_premiacao text NOT NULL DEFAULT '',
  prazo_dias integer,
  ativo boolean NOT NULL DEFAULT true,
  ordem integer NOT NULL DEFAULT 0,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.regras_bonificacao TO anon, authenticated;
GRANT ALL ON public.regras_bonificacao TO service_role;
ALTER TABLE public.regras_bonificacao ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all regras_bonificacao" ON public.regras_bonificacao FOR ALL USING (true) WITH CHECK (true);

-- Reconhecimentos registrados
CREATE TABLE public.reconhecimentos (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tipo_id uuid NOT NULL REFERENCES public.reconhecimento_tipos(id) ON DELETE CASCADE,
  data date NOT NULL,
  projeto_id uuid REFERENCES public.projetos_entradas(id) ON DELETE SET NULL,
  projeto_nome text,
  pessoa_nome text,
  rg_adiantada boolean NOT NULL DEFAULT false,
  projeto_adiantado boolean NOT NULL DEFAULT false,
  observacao text,
  criado_por text,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.reconhecimentos TO anon, authenticated;
GRANT ALL ON public.reconhecimentos TO service_role;
ALTER TABLE public.reconhecimentos ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all reconhecimentos" ON public.reconhecimentos FOR ALL USING (true) WITH CHECK (true);

-- Bonificacoes
CREATE TABLE public.bonificacoes (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  regra_id uuid REFERENCES public.regras_bonificacao(id) ON DELETE SET NULL,
  reconhecimento_id uuid REFERENCES public.reconhecimentos(id) ON DELETE SET NULL,
  marco_chave text NOT NULL UNIQUE,
  marco integer NOT NULL DEFAULT 1,
  projeto_id uuid,
  projeto_nome text,
  pessoa_nome text,
  tipo_nome text NOT NULL DEFAULT '',
  regra_nome text NOT NULL DEFAULT '',
  motivo text NOT NULL DEFAULT '',
  tipo_premiacao text NOT NULL DEFAULT 'financeira',
  valor numeric NOT NULL DEFAULT 0,
  descricao_premiacao text NOT NULL DEFAULT '',
  status text NOT NULL DEFAULT 'disponivel', -- disponivel | entregue | expirada | cancelada
  liberada_em date NOT NULL DEFAULT CURRENT_DATE,
  prazo_limite date,
  entregue_em timestamptz,
  entregue_por text,
  observacao_entrega text,
  cancelada_em timestamptz,
  cancelada_por text,
  justificativa_cancelamento text,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.bonificacoes TO anon, authenticated;
GRANT ALL ON public.bonificacoes TO service_role;
ALTER TABLE public.bonificacoes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all bonificacoes" ON public.bonificacoes FOR ALL USING (true) WITH CHECK (true);

-- Historico de eventos das bonificacoes
CREATE TABLE public.bonificacao_eventos (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  bonificacao_id uuid NOT NULL REFERENCES public.bonificacoes(id) ON DELETE CASCADE,
  evento text NOT NULL,
  detalhe text,
  usuario text,
  created_at timestamptz NOT NULL DEFAULT now()
);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.bonificacao_eventos TO anon, authenticated;
GRANT ALL ON public.bonificacao_eventos TO service_role;
ALTER TABLE public.bonificacao_eventos ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all bonificacao_eventos" ON public.bonificacao_eventos FOR ALL USING (true) WITH CHECK (true);

-- Configuracao geral
CREATE TABLE public.bonificacoes_config (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  prazo_padrao_dias integer,
  updated_at timestamptz NOT NULL DEFAULT now()
);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.bonificacoes_config TO anon, authenticated;
GRANT ALL ON public.bonificacoes_config TO service_role;
ALTER TABLE public.bonificacoes_config ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all bonificacoes_config" ON public.bonificacoes_config FOR ALL USING (true) WITH CHECK (true);

CREATE TRIGGER update_reconhecimento_tipos_updated_at BEFORE UPDATE ON public.reconhecimento_tipos FOR EACH ROW EXECUTE FUNCTION public.update_updated_at_column();
CREATE TRIGGER update_regras_bonificacao_updated_at BEFORE UPDATE ON public.regras_bonificacao FOR EACH ROW EXECUTE FUNCTION public.update_updated_at_column();
CREATE TRIGGER update_reconhecimentos_updated_at BEFORE UPDATE ON public.reconhecimentos FOR EACH ROW EXECUTE FUNCTION public.update_updated_at_column();
