INSERT INTO public.reconhecimento_tipos (nome, escopo, ativo, ordem)
SELECT 'Núcleo de Impacto', 'pessoa', true, COALESCE((SELECT MAX(ordem) FROM public.reconhecimento_tipos), 0) + 1
WHERE NOT EXISTS (SELECT 1 FROM public.reconhecimento_tipos WHERE nome = 'Núcleo de Impacto');

INSERT INTO public.regras_bonificacao (tipo_id, nome, criterio, quantidade_necessaria, ciclo, requer_rg_adiantada, requer_projeto_adiantado, tipo_premiacao, valor, descricao_premiacao, ativo, ordem)
SELECT t.id, 'Núcleo de Impacto', 'por_reconhecimento', 1, 0, false, false, 'financeira', 30, 'R$ 30 por membro do núcleo', true,
  COALESCE((SELECT MAX(ordem) FROM public.regras_bonificacao), 0) + 1
FROM public.reconhecimento_tipos t
WHERE t.nome = 'Núcleo de Impacto'
  AND NOT EXISTS (SELECT 1 FROM public.regras_bonificacao WHERE nome = 'Núcleo de Impacto');
